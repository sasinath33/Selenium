public static void doubleclickMethod(WebDriver driver, Actions act) {

driver.get("https://testautomationpractice.blogspot.com/");

WebElement element2 = driver.findElement(By.id("field1"));

element2.clear();

element2.sendKeys("Techpanda");

}

public static void testAutomation(WebDriver driver, Actions act) {

driver.get("https://testautomationpractice.blogspot.com/");

WebElement element = driver.findElement(By.xpath("//button[@class='dropbtn']"));

act.moveToElement(element).build().perform();

driver.findElement(By.xpath("(//div[@class='dropdown-content']/a)[1]")).click();

}

public static void amazon(WebDriver driver, Actions act) {

driver.get("https://www.amazon.in");

WebElement element = driver.findElement(By.id("nav-link-accountList"));

act.moveToElement(element).build().perform();

List<WebElement> elements = driver.findElements(By.xpath("//div[@id=\"nav-al-your-account\"]/ul/li/a"));

for (int i = 1; i <= elements.size(); i++) {

WebElement element2 = driver.findElement(By.xpath("//div[@id=\"nav-al-your-account\"]/ul/li[" + i + "]/a"));

System.out.println(element2.getText());

}

driver.findElement(By.xpath("//div[@id='nav-al-your-account']/ul/li[12]/a")).click();

}
