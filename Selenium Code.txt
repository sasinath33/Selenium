package demo;

import org.openqa.selenium.By;

import org.openqa.selenium.WebDriver;

import org.openqa.selenium.WebElement;

import org.openqa.selenium.chrome.ChromeDriver;

public class TestAutomation1 {


public static void main(String[] args) {


WebDriver driver=new ChromeDriver();

driver.get("https://testautomationpractice.blogspot.com/");

driver.manage().window().maximize();


WebElement element = driver.findElement(By.cssSelector(".form-control"));

element.sendKeys("TP");

driver.findElement(By.id("email")).sendKeys("techpanda@gmail.com");

driver.findElement(By.id("phone")).sendKeys("987654321");

WebElement element2 = driver.findElement(By.tagName("textarea"));

element2.sendKeys("chennai");

WebElement element3 = driver.findElement(By.name("gender"));

element3.click();


// element.clear();

element.sendKeys("Techpanda");

System.out.println(element.getCssValue("color"));

System.out.println(element.getCssValue("background-color"));

System.out.println(element2.getTagName());

System.out.println(element2.getLocation());

System.out.println(element.getSize());

System.out.println(element.isDisplayed());


System.out.println(element3.isSelected());




System.out.println(driver.getCurrentUrl()+" "+driver.getTitle());

driver.quit();

}

}
